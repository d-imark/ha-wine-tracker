> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Im Bearbeiten-Dialog eine Mehrbild-Galerie (anzeigen/hochladen/löschen/Default wählen) auf Basis der BP1-Endpoints.

**Architecture:** Reine Frontend-Änderung in `_wine_edit_modal.html` (+ `style.css`, `translations.py`). Die Galerie ersetzt im Bearbeiten-Modus den Einzel-Slot; alle Aktionen laufen sofort über die BP1-CRUD-Endpoints.

**Tech Stack:** Jinja/Vanilla JS/CSS. Backend unverändert (BP1). Verifikation per Playwright (System-Chrome, `channel="chrome"`), Bilder via BP1-API vorab angelegt.

## Global Constraints

- Nur Bearbeiten-Modus (Wein hat id); Anlegen behält den Einzel-Slot `#winePreview`.
- Aktionen sofort via `/api/wine/<id>/images…`; `fetch` mit Header `X-Requested-With: XMLHttpRequest`.
- Kein Rotieren/Zuschneiden (YAGNI).
- Kategorien `scan/vivino/ki/manuell`; Upload-Default `manuell`.

// ── Edit-dialog image gallery (BP2) ───────────────────────────────────────────
function initWineGallery(wineId) {
  var gal = document.getElementById('wineGallery');
  var single = document.getElementById('winePreview');
  if (wineId) {
    if (single) single.style.display = 'none';
    if (gal) gal.style.display = '';
    loadWineGallery();
  } else {
    if (gal) gal.style.display = 'none';
    if (single) single.style.display = '';
  }
}

function loadWineGallery() {
  if (!_editWineId) return;
  fetch(INGRESS + '/api/wine/' + _editWineId + '/images',
        { headers: { 'X-Requested-With': 'XMLHttpRequest' } })
    .then(function(r) { return r.json(); })
    .then(function(d) { if (d.ok) renderWineGallery(d.images); })
    .catch(function() {});
}

function renderWineGallery(imgs) {
  var grid = document.getElementById('galGrid');
  if (!grid) return;
  var catLabel = function(c) { return T['img_cat_' + c] || c; };
  if (!imgs.length) {
    grid.innerHTML = '<div class="gal-empty">' + (T.ref_none || '') + '</div>';
    return;
  }
  grid.innerHTML = imgs.map(function(im) {
    var star = im.is_default ? 'mdi-star' : 'mdi-star-outline';
    return '<div class="gal-item' + (im.is_default ? ' is-default' : '') + '">' +
      '<img src="' + INGRESS + '/uploads/' + im.filename + '" alt="" loading="lazy">' +
      '<span class="gal-badge">' + catLabel(im.category) + '</span>' +
      '<button type="button" class="gal-star" title="' + (T.img_default || '') +
        '" onclick="galSetDefault(' + im.id + ')"><i class="mdi ' + star + '"></i></button>' +
      '<button type="button" class="gal-del" onclick="galDelete(' + im.id + ')">' +
        '<i class="mdi mdi-close"></i></button>' +
    '</div>';
  }).join('');
}
```

function galSetDefault(imageId) {
  fetch(INGRESS + '/api/wine/' + _editWineId + '/images/' + imageId + '/default',
        { method: 'POST', headers: { 'X-Requested-With': 'XMLHttpRequest' } })
    .then(function(r) { return r.json(); })
    .then(function(d) { if (d.ok) loadWineGallery(); else showReloadToast(T.ref_save_error, false); })
    .catch(function() { showReloadToast(T.ref_save_error, false); });
}

function galDelete(imageId) {
  if (!window.confirm(T.img_delete_confirm || 'Delete image?')) return;
  fetch(INGRESS + '/api/wine/' + _editWineId + '/images/' + imageId,
        { method: 'DELETE', headers: { 'X-Requested-With': 'XMLHttpRequest' } })
    .then(function(r) { return r.json(); })
    .then(function(d) { if (d.ok) loadWineGallery(); else showReloadToast(T.ref_save_error, false); })
    .catch(function() { showReloadToast(T.ref_save_error, false); });
}

function galUpload(inputEl) {
  if (!inputEl.files || !inputEl.files[0]) return;
  var fd = new FormData();
  fd.append('image', inputEl.files[0]);
  fd.append('category', document.getElementById('galCategory').value);
  fetch(INGRESS + '/api/wine/' + _editWineId + '/images',
        { method: 'POST', body: fd, headers: { 'X-Requested-With': 'XMLHttpRequest' } })
    .then(function(r) { return r.json(); })
    .then(function(d) { if (d.ok) loadWineGallery(); else showReloadToast(T.ref_save_error, false); })
    .catch(function() { showReloadToast(T.ref_save_error, false); })
    .finally(function() { inputEl.value = ''; });
}
```
